cmake_minimum_required(VERSION 3.22.1)
project(unipus_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_FLAGS_DEBUG "-O2 -g0")
set(CMAKE_CXX_FLAGS_DEBUG "-O2 -g0")
get_filename_component(DEPS "${CMAKE_CURRENT_LIST_DIR}/../../../build/native-deps" ABSOLUTE)
if(NOT EXISTS "${DEPS}/whisper/CMakeLists.txt")
    message(FATAL_ERROR "Run python scripts/prepare_android_audio.py before configuring native speech")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
add_subdirectory("${DEPS}/whisper" whisper)
foreach(LIB avutil swresample avcodec avformat)
    add_library(ff_${LIB} SHARED IMPORTED)
    set_target_properties(ff_${LIB} PROPERTIES IMPORTED_LOCATION "${DEPS}/lib/${ANDROID_ABI}/lib${LIB}.so")
endforeach()
add_library(unipusspeech SHARED speech_jni.cpp)
target_include_directories(unipusspeech PRIVATE "${DEPS}/ffmpeg-include")
target_link_libraries(unipusspeech PRIVATE whisper ff_avformat ff_avcodec ff_swresample ff_avutil android log)
target_link_options(unipusspeech PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
